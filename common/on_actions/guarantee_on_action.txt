###############################################################################################################
# RNO Guarantee On Actions
# Detects when RNO guarantees HOL after completing East_Indies_for_Guarantees
# and fires the East Indies guarantee event chain.
#
# HOI4 on_guarantee scope:
#   ROOT = the country issuing the guarantee
#   FROM = the country being guaranteed
###############################################################################################################

on_actions = {

	on_guarantee = {
		effect = {
			# Trigger the East Indies Guarantee chain only when:
			#   1. RNO is the guarantor
			#   2. HOL is the nation being guaranteed
			#   3. RNO has completed the relevant focus
			#   4. The arrangement has not already been set up
			if = {
				limit = {
					tag = RNO
					FROM = { tag = HOL }
					has_completed_focus = East_Indies_for_Guarantees
					NOT = { has_country_flag = RNO_guaranteed_east_indies }
				}
				country_event = {
					id = RNO_Events_East_Indies.5
					days = 1
				}
			}
		}
	}

}
